from pathlib import Path from PIL import Image from bs4 import BeautifulSoup import io, base64 page_path = Path("/mnt/data/retro-fotobox/index.html") img_setup_path = Path("/mnt/data/Retrobox mit Greenscreen und utensilienbox im einsatz.jpg") img_ready_path = Path("/mnt/data/Retrobox betriebsbereit.jpg") for p in [page_path, img_setup_path, img_ready_path]: if not p.exists(): raise FileNotFoundError(p) def to_webp_data_uri(path: Path, max_width: int = 1800, quality: int = 84) -> str: im = Image.open(path).convert("RGB") if im.width > max_width: new_h = round(im.height * max_width / im.width) im = im.resize((max_width, new_h), Image.Resampling.LANCZOS) buf = io.BytesIO() im.save(buf, "WEBP", quality=quality, method=6) return "data:image/webp;base64," + base64.b64encode(buf.getvalue()).decode("ascii") setup_uri = to_webp_data_uri(img_setup_path) ready_uri = to_webp_data_uri(img_ready_path) soup = BeautifulSoup(page_path.read_text(encoding="utf-8"), "html.parser") # Hero: echte Retrobox-Bilder prominent einsetzen hero_imgs = soup.select(".hero-visual img") if len(hero_imgs) >= 3: hero_imgs[0]["src"] = setup_uri hero_imgs[0]["alt"] = "Retro Fotobox mit Greenscreen und Utensilienbox im Einsatz" hero_imgs[1]["src"] = ready_uri hero_imgs[1]["alt"] = "Betriebsbereite Retro Fotobox mit Studioblitz und Sofortdruck" hero_imgs[2]["src"] = setup_uri hero_imgs[2]["alt"] = "Kompletter Aufbau der Retro Fotobox mit Greenscreen" # Hauptabschnitt Retrobox: betriebsbereite Box als Hauptmotiv main_img = soup.select_one("section#retro .feature-image img") if main_img: main_img["src"] = ready_uri main_img["alt"] = "Betriebsbereite Retro Fotobox mit Monitor, Kamera, Drucker und Studioblitz" # Abschnitt Greenscreen: kompletter Aufbau mit Greenscreen + Utensilienbox feature_sections = soup.select(".feature-split .feature-image img") if len(feature_sections) >= 2: feature_sections[1]["src"] = setup_uri feature_sections[1]["alt"] = "Retro Fotobox mit Greenscreen, Utensilienbox und Eventaufbau" # Zusätzliche Bildbeschreibung für Suchmaschinen ergänzen feature_copy = soup.select_one("section#retro .feature-copy") if feature_copy: note = soup.new_tag("p") note["class"] = "image-note" note.string = ( "Auf den Originalbildern sehen Sie den vollständigen Aufbau mit Kamera, Studioblitz, " "Drucker, Bildschirm, Greenscreen und Utensilienbox." ) first_button = feature_copy.find("a") if first_button: first_button.insert_before(note) else: feature_copy.append(note) # Dezente Gestaltung für die Bildbeschreibung ergänzen style_tag = soup.find("style") if style_tag and ".image-note" not in style_tag.string: style_tag.string += """ .image-note{ margin:18px 0 22px; padding:14px 16px; border-left:3px solid var(--acid); background:rgba(255,255,255,.055); color:var(--muted); border-radius:0 12px 12px 0; font-size:.92rem; } """ # Qualitätsprüfung html = "\n" + str(soup) required = [ "Retro Fotobox mit Greenscreen und Utensilienbox im Einsatz", "Betriebsbereite Retro Fotobox mit Monitor, Kamera, Drucker und Studioblitz", "799 €", "400 Bilder", ] for text in required: if text not in html: raise RuntimeError(f"Pflichtinhalt fehlt: {text}") page_path.write_text(html, encoding="utf-8") print(f"Aktualisiert: {page_path}") print(f"Dateigröße: {page_path.stat().st_size / 1024 / 1024:.2f} MB") print("Die beiden Originalbilder sind eingebettet und erscheinen im Hero sowie in den Hauptabschnitten.")
